C++ 运算符重载

Operator Overloading

class Robot {
  int energy;
public:
  Robot(int e) : energy(e) {}

  // 运算符重载 operator+
  Robot operator+(const Robot& other) {
    return Robot(this->energy + other.energy);
  }
};

void mergeRobots() {
  Robot r1(50);
  Robot r2(50);
  Robot r3 = r1 + r2; // 触发重载
}
